Código Pronto para Bot de Telegram (Python - telebot) 
import telebot
import requests

API_KEY = "SUA_CHAVE_AQUI"
GENERATOR_URL = "http://localhost:3000/api/v1/generate"

bot = telebot.TeleBot("SEU_TOKEN_TELEGRAM_BOT")

@bot.message_handler(commands=['comprar', 'gerar'])
def handle_venda(message):
    user_name = message.from_user.first_name
    user_id = str(message.from_user.id)
    user_contact = f"@{message.from_user.username}" if message.from_user.username else user_name

    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Content-Type": "application/json"
    }

    # Passa os dados do cliente para salvar no seu painel de revendedor
    payload = {
        "customer_name": user_name,
        "customer_id": user_id,
        "customer_contact": user_contact,
        "sale_price": 15.00 # ou o valor que você cobra
    }

    response = requests.post(GENERATOR_URL, json=payload, headers=headers)
    
    if response.status_code == 200:
        data = response.json()
        link = data["link"]
        bot.reply_to(message, f"🎉 Olá {user_name}! Seu acesso foi gerado com sucesso:\n🔗 {link}\n\nObrigado pela compra!")
    elif response.status_code == 402:
        bot.reply_to(message, "⚠️ Manutenção momentânea. Tente novamente em instantes.")
    else:
        bot.reply_to(message, "❌ Erro ao gerar link.")

bot.polling()
